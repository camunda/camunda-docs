<MethodEndpoint
  method={"get"}
  path={"/backupRuntime/{backupIdOrPrefix}"}
  context={"endpoint"}
></MethodEndpoint>

Detailed information of the backup with the given backup id, or a list of the backups whose
ids match a prefix ending with '\*'. Covers every physical tenant unless `physicalTenant`
narrows it to one.

<ParamsDetails
  parameters={[
    {
      name: "backupIdOrPrefix",
      required: true,
      in: "path",
      description:
        "A prefix of ids ending with '*', or a full backup id. Defaults to '*', matching all backup ids.",
      schema: {
        oneOf: [
          {
            title: "Backup ID",
            description:
              "The ID of the backup. The ID of the backup must be a positive numerical value. As backups\nare logically ordered by their IDs (ascending), each successive backup must use a higher\nID than the previous one.\n",
            type: "integer",
            format: "int64",
            example: 1,
            minimum: 0,
          },
          {
            title: "Backup ID Prefix",
            description:
              "Prefix of a backup id, followed by an '*' as wildcard to match any backups with a backup ID\nstarting with the given prefix.\n",
            type: "string",
            pattern: "^\\d*\\*$",
            example: "17567*",
          },
        ],
      },
    },
    {
      name: "physicalTenant",
      required: false,
      in: "query",
      description:
        "Narrows the operation to the given physical tenant. When omitted, the operation covers every\nphysical tenant of the cluster, which on a single-tenant cluster is the one `default` tenant\nand therefore unchanged behavior.\n",
      schema: {
        title: "Physical Tenant ID",
        description:
          "The id of a physical tenant, as configured under `camunda.physical-tenants`. A cluster that\nconfigures none has exactly one physical tenant, `default`.\n",
        type: "string",
        example: "tenanta",
      },
    },
  ]}
></ParamsDetails>
